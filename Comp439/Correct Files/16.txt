#include <MyLibrary>;
const float Pi = 3.14;
const int Radius = 7;
var int Area, Circumference;

function CalculateCircle;
newb
    Area := Pi * Radius * Radius;
    Circumference := 2 * Pi * Radius;
endb;

newb
    if (Area > Circumference) 
        cout << Area
    else 
        cout << Circumference;
    
endb
exit
